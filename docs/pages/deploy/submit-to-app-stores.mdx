function balPlan(){

 const load={};S.teachers.forEach(q=>{load[q.id]=0});reqs.forEach(r=>{load[r.tea]=(load[r.tea]||0)+r.n});
 const cur={};reqs.forEach(r=>{cur[r.id]=r.tea});

 const idle=S.teachers.filter(q=>!q.free&&capOf(q)>0&&!reqs.some(r=>r.tea===q.id)).map(q=>q.id);
 const usable=q=>q&&!q.free&&capOf(q)>0;
 const moves=[],before={...load};
 for(let g=0;g<500;g++){
  let best=null;
  reqs.forEach(r=>{
   const from=cur[r.id];
   new Set([...(spec[r.sub]||[]),...idle]).forEach(to=>{
    if(to===from)return;const tq=by(S.teachers,to);if(!usable(tq))return;
    const lf=load[from],lt=load[to],nf=lf-r.n,nt=lt+r.n;
    if(Math.max(nf,nt)>=lf)return;
    if(nt>capOf(tq))return;
    const gain=lf-Math.max(nf,nt);
    if(!best||gain>best.gain||(gain===best.gain&&lf>best.lf))best={r,from,to,gain,lf};
   });
  });
  if(!best)break;
  cur[best.r.id]=best.to;load[best.from]-=best.r.n;load[best.to]+=best.r.n;moves.push({id:best.r.id,from:best.from,to:best.to});
 }
 return{moves,before,after:load};
}
